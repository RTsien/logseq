	  func convIntToStr(num int) string {
	  	p := message.NewPrinter(language.Chinese)
	  	return p.Sprintf("%d", num)
	  }
	  
	  // 计算百分比，并且控制百分比字符总长度
	  func convPercentageToStr(num, total int) string {
	  	p := message.NewPrinter(language.Chinese)
	  	val := float64(0)
	  	if total > 0 {
	  		val = float64(num) / float64(total) * 100
	  	}
	  	return p.Sprintf("%6.2f%%", val)
	  }
	  
	  // 用于根据百分比大小，渲染不通的颜色
	  func getBgColorValue(num, total int) int {
	  	val := 0
	  	if total > 0 {
	  		val = 255 - num*255/total
	  	}
	  	return val
	  }
