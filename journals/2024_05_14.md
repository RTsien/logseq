	- ```lua
	  -- filename: code.lua
	  
	  local threads = {}
	  
	  function setup(thread)
	     table.insert(threads, thread)
	  end
	  
	  function init(args)
	     codes = {}
	  end
	  
	  function response(status, headers, body)
	     codes[status] = (codes[status] or 0) + 1
	  end
	  
	  function done(summary, latency, requests)
	     local total_codes = {}
	     for index, thread in ipairs(threads) do
	        local codes = thread:get("codes")
	           for code, count in pairs(codes) do
	              total_codes[code] = (total_codes[code] or 0) + count
	           end
	     end
	     for code, count in pairs(total_codes) do
	        print(code, count)
	     end
	  end

	- 写个demo连接一下
	  ```go
	  package main
	  
	  import (
	  	"context"
	  	"encoding/json"
	  	"flag"
	  	"fmt"
	  	"log"
	  	"time"
	  
	  	"go.mongodb.org/mongo-driver/bson"
	  	"go.mongodb.org/mongo-driver/event"
	  	"go.mongodb.org/mongo-driver/mongo"
	  	"go.mongodb.org/mongo-driver/mongo/options"
	  )
	  
	  var (
	  	database   = flag.String("database", "test_db", "database name")
	  	collection = flag.String("collection", "test_col", "collection name")
	  	uri        = flag.String("uri", "mongodb://127.0.0.1:27000/?maxPoolSize=100", "uri")
	  )
	  
	  func main() {
	  	flag.Parse()
	  
	  	// 创建一个PoolMonitor，用于监控连接池事件
	  	poolMonitor := &event.PoolMonitor{
	  		Event: func(evt *event.PoolEvent) {
	  			evtBytes, _ := json.Marshal(evt)
	  			evtJson := string(evtBytes)
	  			fmt.Println(evt.Type, evtJson)
	  		},
	  	}
	  	clientOptions := options.Client().ApplyURI(*uri).SetPoolMonitor(poolMonitor)
	  	client, err := mongo.Connect(context.TODO(), clientOptions)
	  	if err != nil {
	  		log.Fatal(err)
	  	}
	  	err = client.Ping(context.TODO(), nil)
	  	if err != nil {
	  		log.Fatal(err)
	  	}
	  	fmt.Println("connection(s) healthy", *database, *collection)
	  
	  	// 打印整个collection数据
	  	collection := client.Database(*database).Collection(*collection)
	  	filter := bson.M{}
	  	ctx, cancel := context.WithTimeout(context.TODO(), time.Second*3)
	  	defer cancel()
	  	cur, err := collection.Find(ctx, filter)
	  	if err != nil {
	  		log.Println(err)
	  		return
	  	}
	  	for cur.Next(ctx) {
	  		var result bson.M
	  		err := cur.Decode(&result)
	  		if err != nil {
	  			log.Println(err)
	  			return
	  		}
	  
	  		fmt.Println(result)
	  	}
	  
	  	if err := cur.Err(); err != nil {
	  		log.Println(err)
	  		return
	  	}
	  
	  	err = client.Disconnect(context.TODO())
	  	if err != nil {
	  		log.Fatal(err)
	  	}
	  	fmt.Println("connection(s) closed", *database, *collection)
	  }
	  ```
	  ```bash
	  go run main.go
	  ```
