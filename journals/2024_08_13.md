		  	sinfo := session.FromContext(ctx)
		  	if sinfo == nil {
		  		return false, errors.New("no session info in ctx")

		  		return ok, nil
		  	case Mode_UNOPEN:

		  	}

		  }

			  func (h *Handler) processTrailer(log *mlog.Logger, s *session.Session, fullMethod string, meta *gatepb.Meta, trailer metadata.MD) {
			  	log.Debugf("process trailer, meta: %v, trailer: %v", meta, trailer)

			  		}

			  			}

			  			s.SetSystemOpenBits(v[0])
			  		case "idip-auth-info-bytes":

			  				log.Warnf("process trailer notify failed, err: %v", err)
			  			}
			  		default:
			  			if !strings.HasPrefix(k, "grpc-") {
			  				if meta.Metadata == nil {
			  					meta.Metadata = make(map[string][]byte)
			  				}
			  				meta.Metadata[k] = []byte(v[0])
			  			}

			  	}

	  // checkSystemOpen 检查系统开放统一接口
	  func (s *Server) checkSystemOpen(ctx context.Context, gid int64) (bool, error) {
	  	// 检查IDIP是否限制
	  	if s.ssapiClient.IdipConfigMgr.IsDisabledByGid(int64(zonepb.SystemID_BountyGame), gid) {
	  		return false, nil
	  	}
	  	// 先从session中获取开放状态，没获取到的话再走自己的逻辑
	  	isOpen, err := systemopenutil.CheckSystemOpen(ctx, int64(zonepb.SystemID_BountyGame))
	  	if err == nil {
	  		return isOpen, nil
	  	}
	  
	  	// 检查禁止玩法
	  	sinfo := session.FromContext(ctx)
	  	if sinfo != nil && len(sinfo.IdipAuthInfoBytes) != 0 {
	  		authInfo, err := authcheck.DecodeAuthInfo(sinfo.IdipAuthInfoBytes)
	  		if err == nil {
	  			params := []*idippb.CheckSystemParam{
	  				{BanType: int32(idippb.BanType_BAN_PLAY_GAME), BanOption: int32(zonepb.SystemID_BountyGame)},
	  			}
	  			results := authcheck.CheckMultiSystem(authInfo, uint32(sinfo.Zone), params)
	  			if results != nil && !results[0].Result {
	  				return false, util.ErrorWithPos(ErrSystemDisabledByIDIP.WithTipsMsg("", map[string]string{
	  					"reason":      results[0].Reason,
	  					"punish_time": strconv.FormatInt(results[0].PunishTime, 10),
	  					"end_time":    time.Unix(results[0].EndTime, 0).Format("2006-01-02 15:04:05"),
	  				}))
	  			}
	  		}
	  	}
	  
	  	// session中没有系统开放信息，尝试从缓存数据中读取
	  	isInit := true
	  	isOpen, err = s.model.GetSystemOpen(ctx, gid)
	  	if err != nil {
	  		if ecode.Is(err, mongo.ErrNoDocuments) {
	  			isInit = false
	  		} else {
	  			return false, util.ErrorWithPos(err)
	  		}
	  	}
	  	if isOpen {
	  		return true, nil
	  	}
	  
	  	if _, err := s.ssapiClient.Zone.IsSystemOpen(ctx, gid, int64(zonepb.SystemID_BountyGame)); err != nil {
	  		return false, util.ErrorWithPos(err)
	  	}
	  	// 玩家数据已经初始化过了的话才能设置数据
	  	if isInit {
	  		if err := s.model.UpdateSystemOpen(ctx, gid, true); err != nil {
	  			return false, util.ErrorWithPos(err)
	  		}
	  	}
	  	return true, nil
	  }
