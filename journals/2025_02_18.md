- #red/actsvr 如何实现玩家锁的：在接口拦截器里取出gid，然后上锁
	- ```go
	  		// 执行 handler 前，对于同一玩家添加分布式锁，避免同一玩家分布式并发
	  		ctx, unlock, err := svr.LockRedisWithRedPointCache(ctx, rediskey.DistributedLockID("act", gid))
	  		if err != nil {
	  			mlog.Ctx(ctx).Warn("distributed lock failed in grpc server interceptor", zap.Error(err))
	  			return nil, util.ErrorWithPos(err)
	  		}
	  		defer func(ctx context.Context, unlock func(context.Context) error) {
	  			if err := unlock(ctx); err != nil {
	  				mlog.Ctx(ctx).Warn("distributed unlock failed in grpc server interceptor", zap.Error(err))
	  			}
	  		}(ctx, unlock)
	  ```
