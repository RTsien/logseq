- #RED/zonesvr #RED/actsvr ProcessGameOver和ProcessGameEvent的区别
	- zonesvr里，核心代码在`zonesvr/internal/pkg/systems/fight/impl/ss_api.go`
	- actsvr里，由各个活动自己实现接口，通常会调用zonesvr里对应的接口
	  ```go
	  // ProcessGameOver 战斗结束
	  func (act *Act) ProcessGameOver(ctx context.Context, msg *corepb.GameOverMessage) (*ssfightpb.ProcessGameOverReply, error) {
	  	gid, err := actpb.GetGidFromCtx(ctx)
	  	if err != nil {
	  		return nil, util.ErrorWithPos(acterr.ErrPlayerNotExist)
	  	}
	  
	  	if err := act.settleGameOverLogics(ctx, msg); err != nil {
	  		mlog.Ctx(ctx).Warnf("player: %v, settle game over message logics failed, err: %v", gid, err)
	  	}
	  
	  	if _, err := act.server.ZoneProcessGameOver(zonepb.WithTargetGID(ctx, gid), msg); err != nil {
	  		return nil, util.ErrorWithPos(err)
	  	}
	  
	  	return &ssfightpb.ProcessGameOverReply{}, nil
	  }
	  
	  // ProcessGameEvent 战斗事件
	  func (act *Act) ProcessGameEvent(ctx context.Context, msg *corepb.GameEventMessage) (*ssfightpb.ProcessGameEventReply, error) {
	  	gid, err := actpb.GetGidFromCtx(ctx)
	  	if err != nil {
	  		return nil, util.ErrorWithPos(err)
	  	}
	  
	  	if _, err := act.server.ZoneProcessGameEvent(zonepb.WithTargetGID(ctx, gid), msg); err != nil {
	  		return nil, util.ErrorWithPos(err)
	  	}
	  
	  	return &ssfightpb.ProcessGameEventReply{}, nil
	  }
	  ```
	- 简单看代码，没办法直观推测出这两个函数各自用于什么场景？特别是到底应该把对局结算的逻辑放到哪一个函数里比较合适。
	- 实际这两个ss函数的client最源头是[[RED/fightsvr]]，具体代码在`internal/pkg/notifier/grpc.go`
		- fightsvr会根据结算通道确定调用到对应svr上：`room, zone, guild, war, act, raid`
