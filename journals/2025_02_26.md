- #RED/zonesvr zonedata实现了*core.Handoff*接口，能够保证zone在迁移之前，zonedata的数据会被回写
	- ```go
	  // OnHandOff implements of core.Handoff
	  func (s *ZoneDataSystemImpl) OnHandOff(ctx context.Context, shardID int) {
	  	// 去卸载小区数据
	  	s.mu.Lock()
	  	defer s.mu.Unlock()
	  
	  	mlog.Ctx(ctx).Infof("zonedata handoff, shardID:%v", shardID)
	  	zd, ok := s.datas[uint32(shardID)]
	  	if !ok { // 如果不存在，就不处理
	  		return
	  	}
	  	err := s.SaveData(ctx, uint32(shardID), true, zd)
	  	if err != nil {
	  		mlog.Ctx(ctx).Error("save data fail", zap.Uint32("zone", uint32(shardID)), zap.Error(err))
	  	}
	  	delete(s.datas, uint32(shardID))
	  }
	  ```
	- 会在shard迁移的时候，在shutdown函数里被执行
	  ```go
	  func (shard *Shard) Shutdown() []uint64 {
	  	// 迁移后，将这个服的在线人数清0
	  	shard.metrics.SetActorCnt(shard.shardID, 0)
	  
	  	log.Infof("[shard shutdown] calling handoffCB")
	  	shard.handoffCB()
	  	log.Infof("[shard shutdown] done handoffCB")
	  
	  	shard.cachedStoppedActorIDsLocked = activeActorIDs
	  	shard.stoppedLocked = true
	  
	  	return activeActorIDs
	  }
	  ```
