- 批量导入数据到[[redis]]
  ```bash
  #!/bin/bash
  
  # Check if redis-cli is available
  if ! command -v redis-cli &> /dev/null; then
      echo "Error: redis-cli is not installed"
      exit 1
  fi
  
  # Check if Redis URI is provided
  if [ -z "$1" ]; then
      echo "Usage: $0 <redis-uri>"
      echo "Example: $0 redis://user:password@localhost:6379/0"
      exit 1
  fi
  
  REDIS_URI="$1"
  KEY_PREFIX="@richie-dev_1207@match_openid_tag_"
  LOG_FILE="redis_commands.log"
  
  # Check if input file exists
  if [ ! -f "kihan_gopenid.txt" ]; then
      echo "Error: kihan_gopenid.txt not found"
      exit 1
  fi
  
  # Initialize counter and buffer
  count=0
  buffer=""
  
  echo "Starting import at $(date)" > "$LOG_FILE"
  
  # Read the file line by line
  while IFS= read -r line; do
      # Skip empty lines and trim whitespace
      line=$(echo "$line" | tr -d '[:space:]')
      [ -z "$line" ] && continue
      
      # Create SET command
      command="SET ${KEY_PREFIX}${line} 1"
      
      # Add command to buffer and log file
      buffer="${buffer}${command}\n"
      echo "$command" >> "$LOG_FILE"
      ((count++))
      
      # When buffer reaches 1000 items, send to Redis
      if [ $count -eq 1000 ]; then
          # Execute redis-cli and suppress OK responses
          echo -e "$buffer" | redis-cli -u "$REDIS_URI" | grep -v "^OK$"
          count=0
          buffer=""
      fi
  done < "kihan_gopenid.txt"
  
  # Process remaining items in buffer
  if [ ! -z "$buffer" ]; then
      # Execute redis-cli and suppress OK responses
      echo -e "$buffer" | redis-cli -u "$REDIS_URI" | grep -v "^OK$"
  fi
  
  echo "Import completed at $(date)" >> "$LOG_FILE"
  echo "Import completed. All commands have been logged to $LOG_FILE"
  
  ```
