	- 物件是有初始状态的，在SeaComponent表里定义。通过model获得的部件信息不会加载物件初始状态
		- ```go
		  func (m *PlayerModel) GetComponentState(ctx context.Context, r *player.Player, componentID int32, componentMapID int64) *entity.SeaComponent {
		  	defaultComponent := &entity.SeaComponent{
		  		Value:          int32(voyageexplore.UnknownComponentValue), // -1是model里没物件记录的初始值
		  		Pos:            voyageexplore.UnknownComponentPos,
		  		Custom:         map[int32]int32{},
		  		Ship:           &entity.ShipComponent{},
		  		SkillIds:       []int64{},
		  		SkillCdStartMs: map[int64]int64{},
		  		BuffStartMs:    map[int64]*entity.Buff{},
		  	}
		    ......
		  }
		  ```
	- 要使用GetComponentValue才行
		- ```go
		  func GetComponentValue(ctx context.Context, r *player.Player, pc *config.ParsedConfig, componentID int64) int64 {
		  	defaultValue := pc.GetComponentInitValue(componentID)
		  	if defaultValue == voyageexplore.UnknownComponentValue {
		  		// 说明都不在配置中，可能是配置错误
		  		mlog.Ctx(ctx).Errorf("GetComponentValue: unknown component: cid=%d", componentID)
		  		return defaultValue
		  	}
		  	state := mustPlayerModel(r).GetComponentState(ctx, r, int32(componentID), pc.GetComponentMapID(componentID))
		  	if state.IsCarried || state.IsDeleted || state.IsFollowing {
		  		return voyageexplore.UnknownComponentValue
		  	}
		  	if state.Value == int32(voyageexplore.UnknownComponentValue) {
		  		return defaultValue
		  	}
		  	return int64(state.Value)
		  }
