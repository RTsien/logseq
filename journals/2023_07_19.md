- 独立集群加入峰峦
	- 准备账号
		- ```yaml
		  ---
		  apiVersion: v1
		  kind: Namespace
		  metadata:
		    annotations:
		      description: 峰峦集群overlay专用
		    name: ten-5-sz-surface-system
		  ---
		  apiVersion: v1
		  kind: ServiceAccount
		  metadata:
		    name: sa-fengluan
		    namespace: default
		  ---
		  apiVersion: rbac.authorization.k8s.io/v1
		  kind: ClusterRole
		  metadata:
		    name: cr-fengluan-readonly
		  rules:
		  - apiGroups:
		    - '*'
		    resources:
		    - '*'
		    verbs:
		    - get
		    - list
		    - watch
		  ---
		  apiVersion: rbac.authorization.k8s.io/v1
		  kind: ClusterRoleBinding
		  metadata:
		    name: crb-fengluan-readonly
		  roleRef:
		    apiGroup: rbac.authorization.k8s.io
		    kind: ClusterRole
		    name: cr-fengluan-readonly
		  subjects:
		  - kind: ServiceAccount
		    name: sa-fengluan
		    namespace: default
		  ---
		  apiVersion: rbac.authorization.k8s.io/v1
		  kind: ClusterRole
		  metadata:
		    name: cr-fengluan-all-perm
		  rules:
		  - apiGroups:
		    - '*'
		    resources:
		    - '*'
		    verbs:
		    - '*'
		  ---
		  apiVersion: rbac.authorization.k8s.io/v1
		  kind: RoleBinding
		  metadata:
		    name: rb-fengluan-all-perm
		    namespace: ten-5-sz-surface-system
		  roleRef:
		    apiGroup: rbac.authorization.k8s.io
		    kind: ClusterRole
		    name: cr-fengluan-all-perm
		  subjects:
		  - kind: ServiceAccount
		    name: sa-fengluan
		    namespace: default
		  ```
	- [[生成kubeconfig]]
		- ```yaml
		  tokenName=$(kubectl get sa sa-fengluan -n default -o "jsonpath={.secrets[0].name}")
		  token=$(kubectl get secret $tokenName -n default -o "jsonpath={.data.token}" | base64 --decode)
		  certificate=$(kubectl get secret $tokenName -n default -o "jsonpath={.data['ca\.crt']}")
		  
		  echo "apiVersion: v1
		  kind: Config
		  preferences: {}
		  clusters:
		  - cluster:
		      certificate-authority-data: $certificate
		      server: ---endpoint---
		    name: default-cluster
		  users:
		  - name: sa-fengluan
		    user:
		      as-user-extra: {}
		      token: $token
		  contexts:
		  - context:
		      cluster: default-cluster
		      namespace: default
		      user: sa-fengluan
		    name: default
		  current-context: default"
		  ```
