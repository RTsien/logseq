		  use dialoguer::{theme::ColorfulTheme, Select};
		  
		  fn main() {
		      // 定义选项列表
		      let options = &["选项1", "选项2", "选项3"];
		  
		      // 创建一个Select实例
		      let selection = Select::with_theme(&ColorfulTheme::default())
		          .with_prompt("请选择一个选项")
		          .default(0)
		          .items(&options[..])
		          .interact()
		          .unwrap();
		  
		      // 输出用户选择的选项
		      println!("您选择了: {}", options[selection]);
		  }
