				  	}

				  	for _, actName := range actNames {

				  		id, _ := strconv.Atoi(actName)
				  		actConfig, ok := a.GetConfig().CommonConfig[int64(id)]

				  			if enable, _ := a.svr.FeatoToggle().Enabled(featoCtx, "disable_act"); enable {
