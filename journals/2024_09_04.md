- #mongodb 批量删除database
	- ```js
	  // delete_all_dbs.js
	  
	  // 获取所有数据库列表
	  var dbs = db.adminCommand('listDatabases').databases;
	  
	  // 系统数据库列表
	  var systemDBs = ['admin', 'config', 'local'];
	  
	  // 遍历所有数据库
	  dbs.forEach(function(database) {
	    var dbName = database.name;
	    
	    // 检查是否为系统数据库
	    if (systemDBs.indexOf(dbName) === -1) {
	      print("尝试删除数据库: " + dbName);
	      try {
	        db.getSiblingDB(dbName).dropDatabase();
	        print("数据库 " + dbName + " 已删除");
	      } catch (e) {
	        print("删除数据库 " + dbName + " 时出错: " + e.message);
	      }
	    } else {
	      print("跳过系统数据库: " + dbName);
	    }
	  });
	  
	  print("操作完成");
	  ```
