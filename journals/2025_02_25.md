	  // SaveMail 保存一封邮件，如果 mail 的 ID 已有且 replace == false，不会修改现有邮件内容
	  // 返回是否创建了新邮件
	  func (s *Service) SaveMail(ctx context.Context, mail *Mail, replace bool) (bool, error) {
	  	if len(mail.FullRewardsList()) != 0 {
	  		mail.HasRewards = true // 有奖励邮件辅助标志位
	  	} else {
	  		mail.Claimed = true // 无奖励邮件默认设 Claimed = true
	  	}
	  
	  	method := "$setOnInsert"
	  	if replace {
	  		method = "$set"
	  	}
	  
	  	opts := options.Update()
	  	opts.SetUpsert(true)
	  	res, err := s.MailColl.UpdateOne(ctx, bson.M{
	  		"to": mail.To,
	  		"id": mail.ID,
	  	}, bson.M{
	  		method: mail,
	  	}, opts)
	  	if err != nil {
	  		return false, util.ErrorWithPos(err)
	  	}
	  
	  	isNew := res.UpsertedCount != 0
	  
	  	if isNew && s.PushFunc != nil {
	  		if err := s.PushFunc(ctx, mail.To, &mailpb.NewMailNtf{
	  			Id:      mail.ID,
	  			Subject: mail.Subject,
	  		}); err != nil {
	  			mlog.Ctx(ctx).Warn("failed to push new mail ntf", zap.Error(err), zap.String("mailid", mail.ID), zap.Int64("to", mail.To))
	  		}
	  	}
	  
	  	return isNew, nil
	  }
