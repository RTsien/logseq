	- ```go
	  // 刷新系统开放状态
	  func (s *SystemOpenSystem) Refresh(ctx context.Context, r *player.Player) {
	  	cache := make(map[int64]bool, len(s.ParsedConfig().SystemOpenConfigs))
	  	opened, closed := make(map[int64]bool), make(map[int64]bool)
	  	model := systemopen.MustModel(r)
	  
	  	for _, cfg := range s.ParsedConfig().SystemOpenConfigs {
	  		openedBefore := model.SystemOpened(cfg.Id)
	  		openedNow := s.isSystemOpen(ctx, r, cfg, cache)
	  		cache[cfg.Id] = openedNow
	  
	  		if openedBefore && !openedNow { // 由开放变为不开放
	  			closed[cfg.Id] = true
	  			model.CloseSystem(cfg.Id)
	  		}
	  		if !openedBefore && openedNow { // 由不开放变为开放
	  			opened[cfg.Id] = true
	  			model.OpenSystem(cfg.Id)
	  		}
	  	}
	  
	  	if len(opened) > 0 || len(closed) > 0 {
	  		// 推送新解锁系统给客户端
	  		ntf := &zonepb.OpenSystems{
	  			List: make([]*zonepb.SystemOpenInfo, 0, len(opened)+len(closed)),
	  		}
	  		for sysID := range opened {
	  			ntf.List = append(ntf.List, &zonepb.SystemOpenInfo{
	  				Id:   int32(sysID),
	  				Open: true,
	  				Show: true,
	  			})
	  		}
	  		for sysID := range closed {
	  			ntf.List = append(ntf.List, &zonepb.SystemOpenInfo{
	  				Id:   int32(sysID),
	  				Open: false,
	  				Show: false,
	  			})
	  		}
	  
	  		// 发送系统开放 notify
	  		if err := system.Core().SendNtf(ctx, r.Gid, ntf); err != nil {
	  			mlog.Ctx(ctx).Warnf("failed to send system open ntf to client: %v", err)
	  		}
	  
	  		// 发出系统开放事件
	  		if err := system.Core().PublishEvent(ctx, r.Gid, &ieventpb.SystemOpenEvent{
	  			OpenedSystems: opened,
	  			ClosedSystems: closed,
	  		}); err != nil {
	  			mlog.Ctx(ctx).Warnf("failed to publish event: %v", err)
	  		}
	  
	  		// 计算系统开放 bitmap，并且推送到 gate
	  		openedSystems, err := s.encodedOpenedSystems(ctx, r)
	  		if err != nil {
	  			mlog.Ctx(ctx).Warnf("get encoded opened system: %v", err)
	  			return
	  		}
	  		if err := system.Core().SendNtf(ctx, r.Gid, &zonepb.SystemOpenChangeNtf{
	  			OpenSystemBits: openedSystems,
	  		}); err != nil {
	  			mlog.Ctx(ctx).Warnf("failed to send system open change ntf to gate: %v", err)
	  		}
	  		// 同步info
	  		infoproxy.MustSystem().MarkDirty(r, s)
	  	}
	  }
